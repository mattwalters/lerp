{{- /* The published files are this repository's own markdown, so their links are written for a reader on GitHub: relative paths to other files in the tree. Left alone they would 404 here — nothing serves SCOPE.md at a site path. Each one is resolved to the page the site publishes for it, or, for a file the site does not publish, to that file on GitHub at the ref this build documents: a frozen /vX.Y.Z/ build links into vX.Y.Z, so it keeps describing the release it was cut from. In-page anchors and absolute URLs are emitted exactly as written; the site paths below start at the root and Hugo's relativeURLs pass makes them relative to the page being rendered. */ -}} {{- $href := .Destination -}} {{- $u := urls.Parse $href -}} {{/* A destination beginning "//" is a URL with the scheme left off, not a path into this repository. */}} {{- if and (not $u.IsAbs) $u.Path (not (hasPrefix $href "//")) -}} {{/* Cleaned and de-slashed first: ./README.md and /README.md are ordinary spellings on GitHub, and both would otherwise miss the page lookup and be sent to a blob URL — one of them a double-slashed 404. */}} {{- $clean := path.Clean $u.Path -}} {{- $path := strings.TrimPrefix "/" $clean -}} {{/* Asked of the page first, then of the site. The mounted files sit at the site root, so for them the two are the same lookup — but a page of the manual sits in a directory, and there the ordinary markdown spelling of a link to its neighbour is a bare filename. Rooted first, that bare name misses every page and leaves for a GitHub blob URL that does not exist, because a Hugo content file is not at that repo path. Nothing that resolved before stops resolving: the rooted lookup is still here, one line down. */}} {{- $page := .Page.GetPage (strings.TrimSuffix ".md" (lower $clean)) -}} {{- if not $page -}} {{- $page = site.GetPage (printf "/%s" (strings.TrimSuffix ".md" (lower $path))) -}} {{- end -}} {{- if $page -}} {{- $href = $page.RelPermalink -}} {{- else -}} {{- $ref := cond (hasPrefix site.Params.version "v") site.Params.version "main" -}} {{- $href = printf "%s/blob/%s/%s" site.Params.repo $ref $path -}} {{- end -}} {{- with $u.RawQuery }}{{ $href = printf "%s?%s" $href . }}{{ end -}} {{- with $u.Fragment }}{{ $href = printf "%s#%s" $href . }}{{ end -}} {{- end -}} {{ .Text | safeHTML }}
