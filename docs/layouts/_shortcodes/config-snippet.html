{{- $name := or (.Get "src") (.Get 0) -}} {{- if not $name -}} {{- errorf "config-snippet: %s missing snippet name (e.g. {{< config-snippet src=\"shape\" >}})" .Position -}} {{- end -}} {{- $tomlPath := printf "snippets/%s.toml" $name -}} {{- $yamlPath := printf "snippets/%s.yaml" $name -}} {{- $jsonPath := printf "snippets/%s.json" $name -}} {{- $toml := strings.TrimRight "\n\r" (readFile $tomlPath) -}} {{- $yaml := strings.TrimRight "\n\r" (readFile $yamlPath) -}} {{- $json := strings.TrimRight "\n\r" (readFile $jsonPath) -}}

{{ highlight $yaml "yaml" "" }}
{{ highlight $json "json" "" }}
